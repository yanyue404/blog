惰性载入函数

在某个场景下我们的函数中有判断语句，这个判断依据在整个项目运行期间一般不会变化，所以判断分支在整个项目运行期间只会运行某个特定分支，那么就可以考虑惰性载入函数

function foo() {
  if (a !== b) {
    console.log("aaa");
  } else {
    console.log("bbb");
  }
}

// 优化后
function foo() {
  if (a != b) {
    foo = function() {
      console.log("aaa");
    };
  } else {
    foo = function() {
      console.log("bbb");
    };
  }
  return foo();
}

第一次运行之后就会覆写这个方法，下一次再运行的时候就不会执行判断了。当然现在只有一个判断，如果判断很多，分支比较复杂，那么节约的资源还是可观的。

举一反三，既然在函数内部定义与原函数同名函数导出可以覆写原函数，在仅需让函数方法执行一次的场景也同样使用，将原函数覆写为空函数就可以了

002. JS-根据条件给对象插入属性

对一个原始类型的值执行扩展运算符，它总会返回一个空对象。比如{ ...true }相当于{}。当然字符串除外，因为字符串拥有某些数组的特性。

那这个奇技淫巧的原理是什么呢？

其实是因为&&运算符的优先级比...高，所以{ ...name && { name } }相当于{ ...(name && { name }) }，这样一转化就好理解了。

扩展运算符的优先级很低，仅高于,。

// 通常我们都这么写

const profile = { age: 18, gender: "男" };

if (name) {
  profile.name = name;
}

// if 可以简化一下

if (name) profile.name = name;

// 你不能这样写，因为逻辑与运算符后面只能跟表达式，不能跟语句

name && profile.name = name;

// 现在你可以这样写

const profile = { age: 18, gender: "男", ...(name && { name }) };

003. Node.js 的 Error-First Callbacks

  • 1.回调函数的第一个参数保留给一个错误 error 对象，如果有错误发生，错误将通过第一个参数 err 返回。
  • 2.回调函数的第二个参数为成功响应的数据保留，如果没有错误发生，err 将被设置为 null, 成功的数据将从第二个参数返回。
Page({
  // 获取用户可以领取的优惠券
  getMyCoupons(callback) {
    fetch._post(API.get_coupons, {}).then(res => {
      let err, result;
      if (res.response_data.error_code) {
        err = res.response_data;
      } else {
        result = res.response_data.lists;
      }
      typeof callback === "function" ? callback(err, result) : "";
    });
  }
});

005. 从数组循环的 forEach 方法中删除元素

let data = [
  {
    name: "蔬菜类",
    thumb_img: "",
    price: 30,
    desc: "茄子",
    goods_id: "4",
    num: 1
  },
  {
    name: "越南进口红心火龙果 4个装",
    thumb_img: "",
    price: 0.01,
    goods_id: "7",
    num: 1
  }
];
let ids = ["4", "7"];
  • 方法一:反向删除 for while
let index = data.length - 1;
while (index >= 0) {
  if (ids.includes(data[index]["goods_id"])) {
    data.splice(index, 1);
  }
  index--;
}
for (var i = data.length - 1; i >= 0; i--) {
  if (ids.includes(data[i]["goods_id"])) {
    data.splice(i, 1);
  }
}
  • 方法二：浅拷贝数组：倒序修正 index 指向
// 使用 slice 浅拷贝 object 是真实的
data.slice().forEach((v, index, object) => {
  ids.includes(v["goods_id"]) ? data.splice(object.length - 1 - index, 1) : "";
});
  • 双重 forEach
data.slice().forEach((v, index, object) => {
  ids.forEach(m => {
    v.goods_id === m ? data.splice(object - 1 - index, 1) : "";
  });
});

class Validator {
  constructor(conditions) {
    setTimeout(() => {
      for (let item of conditions) {
        const res = item.fn();
        if (!res) {
          alert(item.err);
          return;
        }
      }
      this.callback();
    }, 0);
  }
  then(callback = () => {}) {
    this.callback = callback;
  }
}
var vm = new Vue({

007. JS 的加减乘除运算

将浮点数 toString 后记录小数位的长度，然后将小数点抹掉，完整的代码如下：

Math.pow() 函数返回基数（base 例:10）的指数（exponent 例：2）次幂，Math.pow(10,2) = 100

/**
 * 加法
 * @param arg1
 * @param arg2
 * @returns
 **/
function accAdd(arg1, arg2) {
  var r1, r2, m;
  try {
    r1 = arg1.toString().split(".")[1].length;
  } catch (e) {
    r1 = 0;
  }
  try {
    r2 = arg2.toString().split(".")[1].length;
  } catch (e) {
    r2 = 0;
  }
  m = Math.pow(10, Math.max(r1, r2));
  return (arg1 * m + arg2 * m) / m;
}

/**
 * 减法
 * @param arg1
 * @param arg2
 * @returns
 */
function accSub(arg1, arg2) {
  var r1, r2, m, n;
  try {
    r1 = arg1.toString().split(".")[1].length;
  } catch (e) {
    r1 = 0;
  }
  try {
    r2 = arg2.toString().split(".")[1].length;
  } catch (e) {
    r2 = 0;
  }
  m = Math.pow(10, Math.max(r1, r2));
  //动态控制精度长度
  n = r1 >= r2 ? r1 : r2;
  return ((arg1 * m - arg2 * m) / m).toFixed(n);
}

function accMul(arg1, arg2) {
  var m = 0,
    s1 = arg1.toString(),
    s2 = arg2.toString();
  try {
    m += s1.split(".")[1].length;
  } catch (e) {}
  try {
    m += s2.split(".")[1].length;
  } catch (e) {}
  return (
    (Number(s1.replace(".", "")) * Number(s2.replace(".", ""))) /
    Math.pow(10, m)
  );
}

function accDiv(arg1, arg2) {
  var t1 = 0,
    t2 = 0,
    r1,
    r2;
  try {
    t1 = arg1.toString().split(".")[1].length;
  } catch (e) {}
  try {
    t2 = arg2.toString().split(".")[1].length;
  } catch (e) {}
  with (Math) {
    r1 = Number(arg1.toString().replace(".", ""));
    r2 = Number(arg2.toString().replace(".", ""));
    return (r1 / r2) * pow(10, t2 - t1);
  }
}

//给Number类型增加一个add方法，调用起来更加方便。
Number.prototype.add = function(arg) {
  return accAdd(arg, this);
};
Number.prototype.sub = function(arg) {
  return accSub(arg, this);
};
Number.prototype.Mul = function(arg) {
  return accMul(arg, this);
};
Number.prototype.Dev = function(arg) {
  return accMul(arg, this);
};

014. require 自动导入

/**
 * @desc webpack打包入口文件

let moduleExports = {};

const r = require.context('./', true, /^\.\/.+\/.+\.js$/);
r.keys().forEach(key => {
    let attr = key.substring(key.lastIndexOf('/') + 1, key.lastIndexOf('.'));
    moduleExports[attr] = r(key);
});

module.exports = moduleExports;

017. 微信内置浏览器缓存清理

之前做过很多公众号的项目，项目写完后给客户看项目，客户一而再再而三的修改元素向左挪1px，向右挪2px。改好之后让客户看，客户说我特泽发克，你啥都没有修改，你竟然骗我！！！

这其实就是微信内置浏览器的缓存在作祟啦，那么如何清理微信内置浏览器的缓存呢？

你们是否知道 ios版微信 和 android版微信 的内置浏览器的内核是不一样的呢？

android版微信内置浏览器（X5内核）
在安卓版微信内打开链接 http://debugx5.qq.com

拉到调试页面的最底端，勾选上所有的缓存项目，点击清除。

点击确定之后即可完成清除微信浏览器缓存的操作。

ios版微信内置浏览器（WKWebView）
ios版微信内置浏览器内核并不是 X5内核，而是使用的ios的浏览器内核WKWebView，所以安卓手机的那种方案对ios手机用户不生效，因为那个链接压根打不开

只要微信用户退出登录，然后重新登录，ios版微信内置浏览器内核即可清除，不行的话，你们回来打我

有人说了：“IOS中 设置—通用----存储空间 就会看到“正在计算空间”计算完了会清理一点清理即可”，这种办法当然也可以，但是这种办法不光是清理微信内置浏览器的缓存，同时也清理其他的一些数据，比如朋友圈的视频图片和聊天记录等等缓存，而且容易误删某些想留下的数据，对于开发而言，我认为退出重新登录是最好的解决办法。

018. 接口 API 参考文档

一、基本规则

1.1 接⼝规则

协议规则：

描述 说明
传输⽅式 采⽤HTTP、HTTPS
提交⽅式 采⽤POST⽅法提交
请求参数类型 Content-Type: application/json
响应参数类型 Content-Type: application/json
字符编码 统⼀采⽤ UTF-8 字符编码
签名算法 MD5().toUpperCase()

1.2 请求参数公共字段

字段名 变量名 必填 类型 描述
调⽤⽅帐号 appId 是 String(32) 分配给调⽤⽅的帐号 id

1.3 返回参数字段格式

字段名 变量名 必填 类型 描述

较早之前使用js加载组件尝试

在较早之前，用js写过一个直接挂载组件到当前dom上的一个方法，分三步：

  1. 使用 Vue.extend，处理需要用js调用的vue单文件组件，返回该vue组件的一个子类
  2. new对应的组件子类，生成对应的组件实例，并使用 .$mount() 挂载组件，返回对应组件的 vm
  3. 这样可以通过 vm.$el 拿到组件dom，append到当前组件dom里，即可完成加载

具体写法如下

// 假设写好了 showInfo.vue 组件，执行clickShow函数直接显示dialog

import ShowInfo from 'showInfo.vue'
// ...

  const Component = Vue.extend(ShowInfo)

  // 挂载后返回对应组件的vm
  let showInfoVue = new Component().$mount()

  // 将组件vm的dom，append到当前页面
  this.$el.appendChild(showInfoVue.$el)
}

但它有一个缺点，常规调用组件时，我们会向子组件里面传入参数或事件。而这种情况不能向调用的子组件传入参数。下面来看另一种方法

使用js将单文件组件挂载到body的通用方法

我们来看看下面的create方法，其实和上面的方法流程基本一致，只是改变了创建对应vue单文件组件实例的方法。这里用render函数来替代之前的Vue.extend来创建对应组件实例，这样可以通过render函数的createElement函数向子组件内部传参数，传方法等。

// create.js
import Vue from "vue";

export default function create(Component, props) {
  // 先创建实例
  const vm = new Vue({
    render(h) {
      // h就是createElement，它返回VNode
      return h(Component, { props });
    }
  }).$mount();

  // 手动挂载
  document.body.appendChild(vm.$el);

  // 销毁方法
  const comp = vm.$children[0];
  comp.remove = function() {
    document.body.removeChild(vm.$el);
    vm.$destroy();
  };
  return comp;
}

上面的例子中，create函数参数接收一个单文件组件对象，以及在调用组件时需要传递给组件的参数，来看看具体使用例子

上面的例子中，我们引入了 create.js 以及 对应的单文件组件。它还是不够简洁，我们可以再封装一次，只需要调用一个js就搞定。如果我们将它在main.js里面引入并挂载到vue实例属性，那么调用就非常方便了。

下面来看看实现思路，关于render函数createElement的options的配置，参见 createElement 参数 - 深入数据对象

// showDialog/index.js
import Vue from "vue";
import DialogComponent from '@/views/jsDialog/src/index.vue'

let TheDialog = null
export default function showDialog(options) {
  // 如果未移除，先移除
  TheDialog && TheDialog.remove()

  TheDialog = create(DialogComponent, {
    on: {
      // 单文件组件内部可以emit该事件，销毁TheDialog组件
      'close-dialog': () => {
        TheDialog.remove()
      }
    },
    props: {
      // 需要传入的属性，单文件组件需要使用props接收
      title: '标题',
      content: '内容'
    }
    // 其他参数
    ...options
  })

  function create(Component, options) {
    // 先创建实例
    const vm = new Vue({
      render(h) {
        // h就是createElement，它返回VNode
        return h(Component, options);
      }
    }).$mount();

    // 手动挂载
    document.body.appendChild(vm.$el);

    // 销毁方法
    const comp = vm.$children[0];
    comp.remove = function() {
      document.body.removeChild(vm.$el);
      vm.$destroy();
    };
    return comp;
  }
}

使用 CSS 属性支持换行

你这句话非常典型地暴露了 99% 的人踩过的同一个坑：

message: '代码是写出来给\n人看的，附带能在机器上运行。'

然后模板里这样写：

<div class="message">{{ message }}</div>

加了 white-space: pre-wrap 还是不换行，对吧？

根本原因

\n 被你写在了 JavaScript 字符串里，但它根本没有真正变成换行符！

你写成 '给\n人看的'，在 JS 里确实是一个真正的换行符（ASCII 10），理论上没问题。

但绝大多数情况下，开发者其实是这样写的（尤其是从后端接口、JSON、数据库、配置文件里拿的数据）：

// 情况1：最常见的坑！来自接口的字符串里根本不是真正的 \n
message: "代码是写出来给\n人看的，附带能在机器上运行。"

// 情况2：JSON 里被转义成了 \\n
{
  "message": "代码是写出来给\\n人看的，附带能在机器上运行。"
}

// 情况3：模板字符串写错了引号
message: `代码是写出来给\n人看的`   // 这里其实是两个字符 \ 和 n

上面这三种情况，字符串里存的其实是 反斜杠 + n 两个可见字符，而不是真正的换行符。所以你在页面上会直接看到 “给\n人看的” 这几个字，white-space: pre-wrap 再牛也救不了你。

正确的几种写法（任选其一）

// 写法1：直接用真正的换行符（推荐！最直观）
message: '代码是写出来给\n人看的，\n附带能在机器上运行。'

// 写法2：模板字符串里直接敲回车（最推荐！！）
message: `代码是写出来给
人看的，附带能在机器上运行。`

// 写法3：从后端返回的是 \\n 的情况，手动还原
this.message = res.data.message.replace(/\\n/g, '\n')

// 写法4：用正则把可见的 \n 转成真的换行符（救命专用）
this.message = res.data.message.replace(/\\n/g, '\n')

验证你现在的数据到底是不是真换行符（复制下面代码直接在浏览器控制台跑）

// 把你的 item.Q 或 message 粘贴进来测试
const test = "代码是写出来给\\n人看的，附带能在机器上运行。"

console.log('长度:', test.length)
console.log('charCodeAt:', test.charCodeAt(7), test.charCodeAt(8)) // 看第7、8个字符的 ASCII
console.log('包含真的换行符吗:', test.includes('\n'))
console.log('包含 \\n 字符串吗:', test.includes('\\n'))

// 正确的应该输出：
// 包含真的换行符吗: true
// 包含 \\n 字符串吗: false

给你一个万能修复方案（直接抄走就行）

在你的组件里加一个计算属性：

<template>
  <div class="message">{{ realMessage }}</div>
</template>

<script>
export default {
  computed: {
    realMessage() {
      if (!this.item.Q) return ''
      // 把 \\n 变成真正的 \n，同时也支持已经正确的 \n
      return this.item.Q
        .replace(/\\n/g, '\n')     // 先处理转义的 \\n
        .replace(/\\r\\n/g, '\n')  // windows 换行
        .replace(/\\r/g, '\n')
    }
  }
}
</script>

<style>

  white-space: pre-wrap;
  word-wrap: break-word;
}
</style>

white-space: pre-wrap 没问题，问题 99% 出在你的字符串里根本没有真正的 \n，而是两个字符 \ 和 n。

把数据里的 \\n 换成真正的 \n（或直接用模板字符串敲回车），立刻就换行了。试试看，秒好

帮我解释 下面 add(1)(2)(3) 的实现方式

  const add = (a) => (b) => (b ? add(a + b) : a)

  const sum = add(1)(2)(3)

  console.log('sum', sum())

好的，这行代码非常经典，它巧妙地结合了 函数柯里化 (Currying)、闭包 (Closure) 和 递归 (Recursion) 来实现一个可以连续调用的加法函数。

我们来一步一步拆解它。

核心代码

const add = (a) => (b) => (b ? add(a + b) : a);

const sum = add(1)(2)(3);

console.log('sum', sum()); // 输出: sum 6

关键概念解析

  1. 函数柯里化 (Function Currying)

    • 简单来说，就是将一个接收多个参数的函数，转变为一系列只接收一个参数的函数。
    • 原本我们可能会写 add(1, 2, 3)，但柯里化后，我们可以写成 add(1)(2)(3)。
    • add 函数 (a) => ... 接收第一个参数 a，然后返回另一个函数 (b) => ...。这个返回的函数再接收下一个参数 b。这就是柯里化的典型特征。

  2. 闭包 (Closure)

    • 闭包是这个技巧能实现的关键。当 add(a) 执行后，它返回的内部函数 (b) => ... 依然可以访问和“记住”外部函数作用域中的变量 a。
    • 每调用一次，新的 a (即 a + b 的结果) 就会被下一个返回的函数“记住”。

  3. 递归 (Recursion)

    • 当内部函数被调用且传入了参数 b 时（b 是一个“真值”，非 undefined, null, 0 等），它会再次调用 add 函数自身：add(a + b)。这形成了一个递归调用，不断地产生新的、记住了累加和的函数。

  4. 终止条件 (Termination Condition)

    • 递归必须有出口，否则会无限循环。这里的出口是 b ? ... : a 这个三元运算符。
    • 当我们最后一次调用 sum() 时，没有传入任何参数，所以 b 的值是 undefined。
    • 在 JavaScript 中，undefined 是一个“假值”(falsy value)，所以 b ? 的判断为 false。
    • 此时，函数不再执行 add(a + b)，而是执行冒号后面的部分：直接返回变量 a。这个 a 正是前面所有步骤累加起来的结果。

一步步分解 add(1)(2)(3)() 的执行过程

让我们跟着代码走一遍：

第 1 步: add(1)

  1. 调用 add 函数，传入 a = 1。
  2. add 函数执行 (a) => ... 部分。
  3. 它返回一个新的函数：(b) => (b ? add(1 + b) : 1)。
    • 关键：这个新函数通过闭包“记住”了 a 的值是 1。

第 2 步: ... (2) (即 add(1)(2))

  1. 上一步返回的函数被立即调用，并传入 b = 2。
  2. 函数内部执行判断 b ? ...。因为 b 是 2 (真值)，所以条件为 true。
  3. 执行 add(a + b)，即 add(1 + 2)，也就是 add(3)。
  4. 这又是一次对 add 函数的递归调用，这次传入的 a 是 3。
  5. add(3) 的结果是返回又一个全新的函数：(b) => (b ? add(3 + b) : 3)。
    • 关键：这个全新的函数通过闭包“记住”了 a 的值是 3。

第 3 步: ... (3) (即 add(1)(2)(3))

  1. 上一步返回的函数被立即调用，并传入 b = 3。
  2. 函数内部执行判断 b ? ...。因为 b 是 3 (真值)，所以条件为 true。
  3. 执行 add(a + b)，即 add(3 + 3)，也就是 add(6)。
  4. 这又是一次递归调用，传入的 a 是 6。
  5. add(6) 的结果是返回最后一个全新的函数：(b) => (b ? add(6 + b) : 6)。
  6. 这个最终的函数被赋值给了变量 sum。
    • 所以，此时 sum 并不是数字 6，而是一个函数！ sum 的值是 (b) => (b ? add(6 + b) : 6)。

第 4 步: sum()

  1. 我们调用了 sum 函数，并且没有传入任何参数。
  2. 所以 b 的值是 undefined。
  3. 函数内部执行判断 b ? ...。因为 b 是 undefined (假值)，所以条件为 false。
  4. 执行冒号后面的部分：直接返回 a。
  5. 在 sum 函数的闭包里，a 的值是多少？是 6。
  6. 因此，sum() 的调用返回了 6。

第 5 步: console.log('sum', sum())

  1. sum() 执行后得到 6。
  2. 控制台最终打印出 sum 6。

总结

add(1)(2)(3) 的过程并不是在计算，而是在构建函数链。每一次调用 (n) 都会返回一个“升级版”的新函数，这个新函数通过闭包记住了之前所有数字的总和。直到最后一次不带参数的调用 ()，它才触发了终止条件，返回最终的累加结果。

这是一个非常优雅的函数式编程范例。

即使你的项目历史包袱重、到处 store.paymentType = xxx 直接改，Timeline 也会全部捕获（Pinia 官方内置支持 direct mutation 的追踪）。
很多人都只看「Inspector（检查器）」标签里的最终状态，而忽略了 Timeline，这就是你当前的问题根源。

2. 辅助手段：用 $subscribe 打日志（快速定位高频变更）

在你的支付 store 文件里（或 main.ts 创建 store 后）加一段调试代码：

import { usePaymentStore } from '@/stores/payment' // 改成你的路径

const paymentStore = usePaymentStore()

// 只在开发环境生效
if (import.meta.env.DEV) {
  paymentStore.$subscribe((mutation, state) => {
    // 只关心 paymentType 的变更
    const changed = mutation.type === 'direct' ||
                    (mutation.type === 'patch object' && mutation.payload?.paymentType !== undefined)

    if (changed) {
      console.group(`🚨 paymentType 发生变更 → ${state.paymentType}`)
      console.log('变更类型:', mutation.type)
      console.log('完整 mutation:', mutation)
      console.trace('调用栈（可点击展开看具体文件行号）')
      console.groupEnd()
    }
  }, { flush: 'sync' }) // sync 模式让日志更接近变更时刻
}
  • mutation.type 告诉你到底是直接赋值还是 $patch
  • console.trace() 会打印调用栈（虽然不一定 100% 精确到原始赋值行，但结合 Timeline 基本能定位到组件/文件）
  • 跑起来后打开控制台，变更发生时就会高亮打印，瞬间知道是哪个页面/逻辑在乱改

3. 更狠的调试技巧（需要精确调用栈时）

如果你觉得上面还不够准，可以临时用 Proxy 包裹 $state（仅开发环境）：

if (import.meta.env.DEV) {
  const originalState = paymentStore.$state
  paymentStore.$state = new Proxy(originalState, {
    set(target, prop, value) {
      if (prop === 'paymentType') {
        console.group(`🔥 paymentType 被直接赋值 → ${value}`)
        console.trace('精确调用栈')
        console.groupEnd()
        // debugger; // 打开这一行会自动断点调试
      }
      return Reflect.set(target, prop, value)
    }
  })
}

注意：这个 Proxy 只在开发环境加，生产环境要删掉（否则可能影响性能/SSR）。

4. 长期建议（解决历史包袱）

  • 把所有直接 paymentType = xxx 重构到 actions 里（setPaymentType(type)），这样以后 Timeline 里会明确显示是哪个 action 触发的，调用栈也干净。
  • 或者写一个 Pinia plugin 全局拦截 direct mutation 并 warning（防止以后再乱改）。
  • 推荐在团队里加 ESLint 规则或 husky hook，禁止直接修改 store state。

先试 Timeline 吧，99% 的情况它就能让你 5 分钟内定位到问题源头。
用完后把调试代码删掉（或用 import.meta.env.DEV 包起来），项目就干净了。

其他方法结合 vue 的组合式 api 分析可以做到吗，如果我想只是使用 计算属性包裹 paymentType ，然后监听这个计算属性的状态

可以做到，但你想的“简单用计算属性包裹 + 监听这个计算属性”这条路走不通，或者说它几乎起不到你想要的“追踪谁修改了 paymentType”的作用。

下面我直接说明为什么不行，以及在 组合式 API 里真正能比较好用的几种追踪/定位方式（由易到难排序）。

为什么 “computed(() => paymentType) + watch” 基本没用？

// 组件里
const paymentStore = usePaymentStore()
const paymentTypeRef = computed(() => paymentStore.paymentType)

watch(paymentTypeRef, (newVal, oldVal) => {
  console.log('变了', oldVal, '→', newVal)
  // 想在这里 console.trace() 看调用栈？
})

结果通常是：

  • 要么根本不触发（最常见）
  • 要么触发了，但调用栈非常浅（指向 watch 本身那一行），完全看不到是谁真正赋值的

根本原因：

  • Pinia 的 state 是 reactive 对象
  • 直接 paymentStore.paymentType = xxx 这种修改是 reactive set 操作
  • Vue 的 watch / computed 只关心依赖是否被重新评估，它不关心是谁执行了 target[key] = value
  • 所以你拿不到原始赋值点的调用栈

在组合式 API 风格下，目前最实用的几种追踪手段（2025–2026 视角）

排序 方式 能拿到调用栈？ 侵入性 推荐场景 备注
1 Vue DevTools Timeline 是（非常清晰） 无 日常定位、历史包袱项目首选 基本必用
2 store.$subscribe + console.trace 部分（看运气） 低 想快速加日志、看到 mutation 类型 见下文代码
3 在组件里用 storeToRefs + watch 否 低 只想知道“什么时候变了、变到什么值” 适合业务逻辑反应
4 临时 Proxy 劫持 $state 是（最清晰） 中 一定要精确定位“哪一行代码在乱改” 开发专用
5 自定义 Pinia plugin 拦截 mutation 是 高 团队长期维护、想统一加日志/告警/断点 最干净但最重

推荐写法（组合式 API 风格）

方案 A：最实用 - $subscribe + trace（放在 main 或插件中）

// src/stores/payment.ts  或新建一个 dev-only 插件
import { storeToRefs } from 'pinia'
import { watch } from 'vue'

export const usePaymentStore = defineStore('payment', () => {
  const state = reactive({
    paymentType: 'wechat' as string,
    // ...
  })

  // 只在开发环境加监控
  if (import.meta.env.DEV) {
    const { paymentType } = storeToRefs(usePaymentStore()) // 这样拿的是 ref

    watch(paymentType, (newV, oldV) => {
      console.groupCollapsed(
        `%c paymentType 变更 %c ${oldV} → ${newV}`,
        'background:#f87171;color:white;padding:2px 6px;border-radius:3px;',
        'color:#f87171;'
      )
      console.log('当前完整 state:', { ...state })
      console.trace('变更来源（点开看调用栈）')
      console.groupEnd()
    }, { flush: 'sync' })  // sync 让日志更接近实际变更时机
  }

  return { ...state /* 其他 getter/action */ }
})

方案 B：最精准 - 临时用 Proxy 劫持（开发环境专用）

if (import.meta.env.DEV) {
  const rawState = paymentStore.$state
  paymentStore.$state = new Proxy(rawState, {
    set(target: any, key: string | symbol, value: any, receiver: any) {
      if (key === 'paymentType') {
        console.groupCollapsed(
          `%c 🔥 paymentType 直接被赋值 → ${value}`,
          'background:#dc2626;color:white;padding:3px 8px;'
        )
        console.trace('赋值调用栈（应该能看到具体组件/工具函数行号）')
        // debugger;   // 想断点就打开这一行
        console.groupEnd()
      }
      return Reflect.set(target, key, value, receiver)
    }
  })
}

这种方式调用栈最干净，经常能直接定位到是哪个页面、哪个 utils、哪个 composable 在直接写 store.paymentType = xxx。

方案 C：如果你只是想“业务上响应变化”，而不是定位bug

那确实可以用 storeToRefs + watch：

<script setup>
import { storeToRefs } from 'pinia'
import { watch } from 'vue'

const paymentStore = usePaymentStore()
const { paymentType } = storeToRefs(paymentStore)

watch(paymentType, (newType) => {
  if (newType === 'alipay') {
    // 做一些副作用，比如切换 UI、发埋点等
  }
})
</script>

但这不适合用来定位“谁在乱改”。

总结一句话建议

  • 先用 Vue DevTools Timeline（零成本、最高效）
  • 如果还定位不到，再加 Proxy 劫持 或 $subscribe + trace（5-10 分钟就能看到元凶）
  • 计算属性 + watch 这条路基本是死胡同，不要浪费时间

你现在的项目历史包袱重，90% 的情况用上面 1 + 4 就能在几分钟内揪出直接赋值的那行代码。
有需要我可以帮你把 Proxy / plugin 写成可复用的 dev 插件形式。
